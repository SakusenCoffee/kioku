cmake_minimum_required(VERSION 3.24)
project(KiokuWebProbe LANGUAGES C CXX)

find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network Svg)

# the Kioku repository (github.com/krmanik/kioku): PyQt6's sip module source
# and the shared kioku Python package; see tools/env.sh
set(KIOKU_REPO "${CMAKE_SOURCE_DIR}/.." CACHE PATH "Kioku repository")
set(DEPS ${CMAKE_SOURCE_DIR}/deps)
set(CPY ${DEPS}/Python-3.14.0)
set(CPY_BUILD ${CPY}/cross-build/wasm32-emscripten/build/python)
set(PYTHON_INCLUDE ${CPY}/Include ${CPY_BUILD})

# Qt's prebuilt wasm libraries use Emscripten's JS setjmp/longjmp, which
# can't be mixed with wasm exception handling, so Rust is built with
# panic=abort (deps/rsbuild.sh) and nothing here uses -fwasm-exceptions.

# generated by tools/gen_pyqt.sh
include(${CMAKE_SOURCE_DIR}/build/pyqt6.cmake)

file(GLOB SIP_SOURCES ${KIOKU_REPO}/ios/deps/sdist/pyqt6_sip-13.13.0/*.c)
add_library(pyqt6_sip STATIC ${SIP_SOURCES})
target_include_directories(pyqt6_sip PRIVATE ${PYTHON_INCLUDE})
target_compile_options(pyqt6_sip PRIVATE -w)

foreach(m QtCore QtGui QtWidgets QtNetwork QtSvg)
    target_include_directories(pyqt6_${m} PRIVATE ${PYTHON_INCLUDE})
    target_compile_options(pyqt6_${m} PRIVATE -w)
endforeach()

qt_add_executable(kioku src/main.cpp)
target_include_directories(kioku PRIVATE ${PYTHON_INCLUDE})
file(GLOB HACL_OBJECTS ${CPY_BUILD}/Modules/_hacl/*.o)
set(PREFIX ${CPY}/cross-build/wasm32-emscripten/prefix)
target_link_libraries(kioku PRIVATE
    pyqt6_QtCore pyqt6_QtGui pyqt6_QtWidgets pyqt6_QtNetwork pyqt6_QtSvg
    pyqt6_sip
    ${CMAKE_SOURCE_DIR}/build/rust-target/wasm32-unknown-emscripten/release/librsbridge.a
    ${CMAKE_SOURCE_DIR}/build/rust-target-rpds/wasm32-unknown-emscripten/release/librpds.a
    ${CPY_BUILD}/libpython3.14.a ${CPY_BUILD}/Modules/expat/libexpat.a ${HACL_OBJECTS}
    ${PREFIX}/lib/libmpdec.a ${PREFIX}/lib/libffi.a ${PREFIX}/lib/libssl.a ${PREFIX}/lib/libcrypto.a
    Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Svg
)
option(KIOKU_ASYNCIFY "Link with Asyncify (nested event loops)" OFF)
if(KIOKU_ASYNCIFY)
    target_link_options(kioku PRIVATE -sASYNCIFY -sASYNCIFY_STACK_SIZE=1048576)
endif()
# Python's Thread.start() waits for the thread to run, which on the main
# thread deadlocks once Emscripten has to create a new Web Worker for it
# (the worker can only start when the main thread yields): pre-create enough
set_target_properties(kioku PROPERTIES QT_WASM_PTHREAD_POOL_SIZE 32)
target_link_options(kioku PRIVATE
    -sUSE_ZLIB -sUSE_BZIP2 -sALLOW_TABLE_GROWTH --profiling-funcs -sSTACK_SIZE=5MB -sFORCE_FILESYSTEM -lidbfs.js -Wl,--wrap=poll
    "SHELL:--pre-js ${CMAKE_SOURCE_DIR}/shell/storage.js"
    "SHELL:--pre-js ${CMAKE_BINARY_DIR}/app-data.js"
    "SHELL:--preload-file ${CMAKE_SOURCE_DIR}/build/stdlib@/python/lib/python3.14"
    "SHELL:--preload-file ${CMAKE_SOURCE_DIR}/build/site@/site"
    "SHELL:--preload-file ${CMAKE_SOURCE_DIR}/deps/fonts@/fonts"
    "SHELL:--exclude-file *__pycache__*"
)

# The app's own Python (app.data) and the page files (shell/), packed on
# every build without relinking (tools/pack_app.sh); kioku.data keeps the
# stdlib, site-packages and fonts
add_custom_target(app_files ALL
    COMMAND ${CMAKE_SOURCE_DIR}/tools/pack_app.sh ${CMAKE_BINARY_DIR}
    BYPRODUCTS ${CMAKE_BINARY_DIR}/app-data.js
    VERBATIM)
add_dependencies(kioku app_files)
set_property(TARGET kioku APPEND PROPERTY LINK_DEPENDS
    ${CMAKE_SOURCE_DIR}/shell/storage.js ${CMAKE_BINARY_DIR}/app-data.js)
